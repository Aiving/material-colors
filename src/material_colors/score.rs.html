15    hct: Hct,
16    score: f64,

19/// Given a large set of colors, remove colors that are unsuitable for a UI
20/// theme, and rank the rest based on suitability.
21///
22/// Enables use of a high cluster count for image quantization, thus ensuring
23/// colors aren't muddied, while curating the high cluster count to a much
24///  smaller number of appropriate choices.
25pub struct Score;
26
27impl Score {
28    const CUTOFF_CHROMA: f64 = 5.0;
29    const CUTOFF_EXCITED_PROPORTION: f64 = 0.01;
30    const TARGET_CHROMA: f64 = 48.0;
31    const WEIGHT_CHROMA_ABOVE: f64 = 0.3;
32    const WEIGHT_CHROMA_BELOW: f64 = 0.1;
33    // A1 Chroma
34    const WEIGHT_PROPORTION: f64 = 0.7;
35
36    /// Given a map with keys of colors and values of how often the color
37    /// appears, rank the colors based on suitability for being used for a
38    /// UI theme.
39    ///
40    /// - Parameters: `colorsToPopulation`: is a map with keys of colors and
41    ///   values of often the color appears, usually from a source image.
42    ///   `desired`: Max count of colors to be returned in the list.
43    ///   `fallbackColorRgb`: Color to be returned if no other options
44    ///   available. `filter`: Whether to filter out undesirable combinations.
45    ///
46    /// - Returns: A list of color `Int` that can be used when generating a
47    ///   theme. The list returned is of length <= `desired`. The recommended
48    ///   color is the first item, the least suitable is the last. There will
49    ///   always be at least one color returned. If all the input colors were
50    ///   not suitable for a theme, a default fallback color will be provided,
51    ///   Google Blue. The default number of colors returned is 4, simply
52    ///   because thats the # of colors display in Android 12's wallpaper
53    ///   picker.
54    pub fn score(colors_to_population: &IndexMap<Rgb, u32>, desired: Option<usize>, fallback_color_rgb: Option<Rgb>, filter: Option<bool>) -> Vec<Rgb> {
55        let desired = desired.unwrap_or(4);
56        let fallback_color_rgb = fallback_color_rgb.unwrap_or(Rgb::new(66, 133, 244));
57        let filter = filter.unwrap_or(true);
58        // Get the HCT color for each Rgb value, while finding the per hue count
59        // and total count.
60        let mut colors_hct = vec![];
61        let mut hue_population = [0; 360];
62        let mut population_sum = 0.0;
63
64        for (rgb, population) in colors_to_population {
65            let hct: Hct = (*rgb).into();
66
67            let hue = hct.get_hue().floor() as i32;
68
69            colors_hct.push(hct);
70
71            hue_population[hue as usize] += population;
72            population_sum += f64::from(*population);
73        }
74
75        // Hues with more usage in neighboring 30 degree slice get a larger
76        // number.
77        let mut hue_excited_proportions = [0.0; 360];
78
79        for (hue, population) in hue_population.into_iter().enumerate().take(360) {
80            let proportion = f64::from(population) / population_sum;
81
82            for i in ((hue as i32) - 14)..((hue as i32) + 16) {
83                let neighbor_hue = sanitize_degrees_int(i);
84
85                hue_excited_proportions[neighbor_hue as usize] += proportion;
86            }
87        }
88
89        // Scores each HCT color based on usage and chroma, while optionally
90        // filtering out values that do not have enough chroma or usage.
91        let mut scored_hcts = vec![];
92
93        for hct in colors_hct {
94            let hue = hct.get_hue().round() as i32;
95
96            let hue = sanitize_degrees_int(hue);
97            let proportion = hue_excited_proportions[hue as usize];
98
99            if filter && (hct.get_chroma() < Self::CUTOFF_CHROMA || proportion <= Self::CUTOFF_EXCITED_PROPORTION) {
100                continue;
101            }
102
103            let proportion_score = proportion * 100.0 * Self::WEIGHT_PROPORTION;
104            let chroma_weight = if hct.get_chroma() < Self::TARGET_CHROMA {
105                Self::WEIGHT_CHROMA_BELOW
106            } else {
107                Self::WEIGHT_CHROMA_ABOVE
108            };
109            let chroma_score = (hct.get_chroma() - Self::TARGET_CHROMA) * chroma_weight;
110            let score = proportion_score + chroma_score;
111
112            scored_hcts.push(ScoredHCT { hct, score });
113        }
114
115        // Sorted so that colors with higher scores come first.
116        // SAFETY: The score will never be NAN, so using `unwrap_unchecked` is
117        // completely safe
118        scored_hcts.sort_by(|a, b| unsafe { b.score.partial_cmp(&a.score).unwrap_unchecked() });
119
120        // Iterates through potential hue differences in degrees in order to
121        // select the colors with the largest distribution of hues
122        // possible. Starting at 90 degrees(maximum difference for 4
123        // colors) then decreasing down to a 15 degree minimum.
124        let mut chosen_colors: Vec<Hct> = vec![];
125
126        for difference_degree in (15..=90).rev() {
127            chosen_colors.clear();
128
129            for entry in &scored_hcts {
130                let hct = entry.hct;
131
132                if !chosen_colors
133                    .iter()
134                    .any(|color| difference_degrees(entry.hct.get_hue(), color.get_hue()) < f64::from(difference_degree))
135                {
136                    chosen_colors.push(hct);
137                }
138
139                if chosen_colors.len() >= desired {
140                    break;
141                }
142            }
143
144            if chosen_colors.len() >= desired {
145                break;
146            }
147        }
148
149        let mut colors = vec![];
150
151        if chosen_colors.is_empty() {
152            colors.push(fallback_color_rgb);
153        }
154
155        for chosen_hct in chosen_colors {
156            colors.push(Rgb::from(chosen_hct));
157        }
158
159        colors
160    }
161}
