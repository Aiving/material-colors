6/// Returns a contrast ratio, which ranges from 1 to 21.
7///
8/// - `toneA`: Tone between 0 and 100. Values outside will be clamped.
9/// - `toneB`: Tone between 0 and 100. Values outside will be clamped.
10pub const fn ratio_of_tones(tone_a: f64, tone_b: f64) -> f64 {
11    let tone_a = tone_a.clamp(0.0, 100.0);
12    let tone_b = tone_b.clamp(0.0, 100.0);
13
14    ratio_of_ys(y_from_lstar(tone_a), y_from_lstar(tone_b))
15}
16
17const fn ratio_of_ys(y1: f64, y2: f64) -> f64 {
18    let lighter = if y1 > y2 { y1 } else { y2 };
19    let darker = if (lighter - y2).abs() < f64::EPSILON { y1 } else { y2 };
20
21    (lighter + 5.0) / (darker + 5.0)
22}

24/// Returns a tone >= `tone` that ensures `ratio`. Return value is between 0 and
25/// 100. Returns -1 if `ratio` cannot be achieved with `tone`.
26///
27/// - `tone`: Tone return value must contrast with. Range is 0 to 100. Invalid
28///   values will result in -1 being returned.
29/// - `ratio`: Contrast ratio of return value and `tone`. Range is 1 to 21,
30///   invalid values have undefined behavior.
31pub fn lighter(tone: f64, ratio: f64) -> Option<f64> {
32    if !(0.0..=100.0).contains(&tone) {
33        return None;
34    }
35
36    let dark_y = y_from_lstar(tone);
37    let light_y = ratio.mul_add(dark_y + 5.0, -5.0);
38
39    if !(0.0..=100.0).contains(&light_y) {
40        return None;
41    }
42
43    let real_contrast = ratio_of_ys(light_y, dark_y);
44    let delta = (real_contrast - ratio).abs();
45
46    if real_contrast < ratio && delta > 0.04 {
47        return None;
48    }
49
50    // Ensure gamut mapping, which requires a 'range' on tone, will still result
51    // the correct ratio by darkening slightly.
52    let return_value = lstar_from_y(light_y) + 0.4;
53
54    if (0.0..=100.0).contains(&return_value) { Some(return_value) } else { None }
55}

57/// Returns a tone <= `tone` that ensures `ratio`. Return value is between 0 and
58/// 100. Returns -1 if `ratio` cannot be achieved with `tone`.
59///
60/// - `tone`: Tone return value must contrast with. Range is 0 to 100. Invalid
61///   values will result in -1 being returned.
62/// - `ratio`: Contrast ratio of return value and `tone`. Range is 1 to 21,
63///   invalid values have undefined behavior.
64pub fn darker(tone: f64, ratio: f64) -> Option<f64> {
65    if !(0.0..=100.0).contains(&tone) {
66        return None;
67    }
68
69    let light_y = y_from_lstar(tone);
70    let dark_y = ((light_y + 5.0) / ratio) - 5.0;
71
72    if !(0.0..=100.0).contains(&dark_y) {
73        return None;
74    }
75
76    let real_contrast = ratio_of_ys(light_y, dark_y);
77
78    let delta = (real_contrast - ratio).abs();
79
80    if real_contrast < ratio && delta > 0.04 {
81        return None;
82    }
83
84    // Ensure gamut mapping, which requires a 'range' on tone, will still result
85    // the correct ratio by darkening slightly.
86    let return_value = lstar_from_y(dark_y) - 0.4;
87
88    if (0.0..=100.0).contains(&return_value) { Some(return_value) } else { None }
89}

91/// Returns a tone >= `tone` that ensures `ratio`. Return value is between 0 and
92/// 100. Returns 100 if `ratio` cannot be achieved with `tone`.
93///
94/// This method is unsafe because the returned value is guaranteed to be in
95/// bounds for tone, i.e. between 0 and 100. However, that value may not reach
96/// the `ratio` with `tone`. For example, there is no color lighter than T100.
97///
98/// - `tone`: Tone return value must contrast with. Range is 0 to 100. Invalid
99///   values will result in 100 being returned.
100/// - `ratio`: Desired contrast ratio of return value and tone parameter. Range
101///   is 1 to 21, invalid values have undefined behavior.
102pub fn lighter_unsafe(tone: f64, ratio: f64) -> f64 {
103    lighter(tone, ratio).unwrap_or(100.0)
104}
