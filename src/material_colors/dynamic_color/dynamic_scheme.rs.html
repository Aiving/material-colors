48/// Constructed by a set of values representing the current UI state and
49/// provides a set of [`TonalPalette`]s that can create colors that fit in with
50/// the theme style.
51///
52/// Used by [`DynamicColor`] to resolve into a color.
53///
54/// At most two source colors are used, so they are stored as
55/// `source_color_hct` + `secondary_source_color_hct` to stay allocation-free.
56#[derive(Clone, PartialOrd)]
57pub struct DynamicScheme {
58    /// The source color of the scheme in HCT format.
59    pub source_color_hct: Hct,
60
61    /// Optional second source color, used by the
62    /// 2026 spec for tertiary roles.
63    pub secondary_source_color_hct: Option<Hct>,
64
65    /// The variant of the scheme.
66    pub variant: Variant,
67
68    /// Whether or not the scheme is dark mode.
69    pub is_dark: bool,
70
71    /// Value from -1 to 1. -1 represents minimum contrast, 0 represents
72    /// standard (i.e. the design as spec'd), and 1 represents maximum contrast.
73    pub contrast_level: f64,
74
75    pub platform: Platform,
76
77    /// Always the *effective* version (after [`Self::fallback_spec_version`]);
78    /// set it through [`Self::with_spec_version`].
79    pub spec_version: SpecVersion,
80
81    pub primary_palette: TonalPalette,
82    pub secondary_palette: TonalPalette,
83    pub tertiary_palette: TonalPalette,
84    pub neutral_palette: TonalPalette,
85    pub neutral_variant_palette: TonalPalette,
86    pub error_palette: TonalPalette,
87}

120    /// Builds a scheme whose palettes come from the given spec.
121    ///
122    /// The effective spec version follows [`Self::fallback_spec_version`]
123    /// (e.g. `Content` is always 2021). `Variant::Cmf` is delegated to
124    /// [`SchemeCmf`], which only exists for spec 2026.
125    pub fn from_spec(
126        source_color_hct: Hct,
127        variant: Variant,
128        is_dark: bool,
129        contrast_level: Option<f64>,
130        platform: Platform,
131        spec_version: SpecVersion,
132    ) -> Self {
133        if variant == Variant::Cmf {
134            return SchemeCmf::with_sources(source_color_hct, None, is_dark, contrast_level, platform).scheme;
135        }
136
137        let contrast = contrast_level.unwrap_or(0.0);
138        let palette = |palette: Palette| Self::spec_palette(variant, source_color_hct, &palette, is_dark, platform, contrast, spec_version);
139
140        Self {
141            source_color_hct,
142            secondary_source_color_hct: None,
143            variant,
144            is_dark,
145            contrast_level: contrast,
146            platform,
147            spec_version: Self::fallback_spec_version(spec_version, variant),
148            primary_palette: palette(Palette::Primary),
149            secondary_palette: palette(Palette::Secondary),
150            tertiary_palette: palette(Palette::Tertiary),
151            neutral_palette: palette(Palette::Neutral),
152            neutral_variant_palette: palette(Palette::NeutralVariant),
153            error_palette: palette(Palette::Error),
154        }
155    }
156
157    /// One palette of a scheme, from `color_spec(spec_version)`,
158    /// with `SchemeCmf`'s palettes for `Variant::Cmf`.
159    pub fn spec_palette(
160        variant: Variant,
161        source_color_hct: Hct,
162        palette: &Palette,
163        is_dark: bool,
164        platform: Platform,
165        contrast_level: f64,
166        spec_version: SpecVersion,
167    ) -> TonalPalette {
168        if variant == Variant::Cmf {
169            return SchemeCmf::palette(&source_color_hct, palette);
170        }
171
172        let spec = color_spec(Self::fallback_spec_version(spec_version, variant));
173        let palette = match palette {
174            Palette::Primary => spec.get_primary_palette(variant, source_color_hct, is_dark, platform, contrast_level),
175            Palette::Secondary => spec.get_secondary_palette(variant, source_color_hct, is_dark, platform, contrast_level),
176            Palette::Tertiary => spec.get_tertiary_palette(variant, source_color_hct, is_dark, platform, contrast_level),
177            Palette::Neutral => spec.get_neutral_palette(variant, source_color_hct, is_dark, platform, contrast_level),
178            Palette::NeutralVariant => spec.get_neutral_variant_palette(variant, source_color_hct, is_dark, platform, contrast_level),
179            Palette::Error => spec.get_error_palette(variant, source_color_hct, is_dark, platform, contrast_level),
180        };
181
182        // Every spec returns `None` only for CMF, handled above.
183        palette.unwrap_or_else(|| unreachable!("fallback spec has no palettes for {variant:?}"))
184    }

215            Variant::TonalSpot => SchemeTonalSpot::new(source_hct, is_dark, contrast_level).scheme,

217            Variant::Expressive => SchemeExpressive::new(source_hct, is_dark, contrast_level).scheme,
218            Variant::Fidelity => SchemeFidelity::new(source_hct, is_dark, contrast_level).scheme,
219            Variant::Content => SchemeContent::new(source_hct, is_dark, contrast_level).scheme,

254    /// Returns the spec version to use for `variant`, falling back when the
255    /// variant is not supported by `spec_version`.
256    pub const fn fallback_spec_version(spec_version: SpecVersion, variant: Variant) -> SpecVersion {
257        match variant {
258            Variant::Cmf => spec_version,
259            Variant::Expressive | Variant::Vibrant | Variant::TonalSpot | Variant::Neutral => match spec_version {
260                SpecVersion::Spec2026 => SpecVersion::Spec2025,
261                v => v,
262            },
263            _ => SpecVersion::Spec2021,
264        }
265    }

267    pub fn get_piecewise_value(source_hue: f64, hue_breakpoints: &[f64], hues: &[f64]) -> f64 {
268        let size = hue_breakpoints.len().saturating_sub(1).min(hues.len());
269
270        for i in 0..size {
271            if source_hue >= hue_breakpoints[i] && source_hue < hue_breakpoints[i + 1] {
272                return sanitize_degrees(hues[i]);
273            }
274        }
275
276        // No condition matched, return the source value.
277        source_hue
278    }
279
280    /// `source_hue` rotated by the value of its breakpoint segment.
281    pub fn get_rotated_hue(source_hue: f64, hue_breakpoints: &[f64], rotations: &[f64]) -> f64 {
282        let rotation = if hue_breakpoints.len().saturating_sub(1).min(rotations.len()) == 0 {
283            // No condition matched, return the source hue.
284            0.0
285        } else {
286            Self::get_piecewise_value(source_hue, hue_breakpoints, rotations)
287        };
288
289        sanitize_degrees(source_hue + rotation)
290    }

306#[inline]
307pub(crate) fn sanitize_degrees(degrees: f64) -> f64 {
308    let degrees = degrees % 360.0;
309
310    if degrees < 0.0 { degrees + 360.0 } else { degrees }
311}

393impl PartialEq for DynamicScheme {
394    fn eq(&self, other: &Self) -> bool {
395        self.source_color_hct == other.source_color_hct
396            && self.secondary_source_color_hct == other.secondary_source_color_hct
397            && self.variant == other.variant
398            && self.is_dark == other.is_dark
399            && self.contrast_level == other.contrast_level
400            && self.platform == other.platform
401            && self.spec_version == other.spec_version
402            && self.primary_palette == other.primary_palette
403            && self.secondary_palette == other.secondary_palette
404            && self.tertiary_palette == other.tertiary_palette
405            && self.neutral_palette == other.neutral_palette
406            && self.neutral_variant_palette == other.neutral_variant_palette
407            && self.error_palette == other.error_palette
408    }
