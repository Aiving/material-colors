3/// Describes how to fulfill a tone delta pair constraint.
4#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
5pub enum DeltaConstraint {
6    /// The tone of `role_a` must be an exact delta away from the tone of
7    /// `role_b`.
8    Exact,
9    /// The tonal distance of `role_a` and `role_b` must be at most delta.
10    Nearer,
11    /// The tonal distance of `role_a` and `role_b` must be at least delta.
12    Farther,
13}
14
15/// Describes the relationship in lightness between two colors.
16///
17/// `RelativeDarker`/`RelativeLighter` are relative to the surface trend (white
18/// in light mode, black in dark mode).
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
20pub enum TonePolarity {
21    Darker,
22    Lighter,
23    RelativeDarker,
24    RelativeLighter,
25}
26
27/// Documents a constraint between two `DynamicColor`s, in which their tones
28/// must have a certain distance from each other.
29///
30/// `polarity` describes `role_a` compared to `role_b`: `ToneDeltaPair(A, B, 15,
31/// Darker)` states that A's tone should be at least 15 darker than B's.
32///
33/// Prefer a `DynamicColor` with a background, this is for special cases when
34/// designers want tonal distance, literally contrast, between two colors that
35/// don't have a background / foreground relationship or a contrast guarantee.
36#[derive(Debug, Clone, Copy)]
37pub struct ToneDeltaPair<'a> {
38    pub role_a: DynamicColor<'a>,
39    pub role_b: DynamicColor<'a>,
40    /// Required difference between tones. Absolute value, negative values
41    /// have undefined behavior.
42    pub delta: f64,
43    pub polarity: TonePolarity,
44    /// Whether these two roles should stay on the same side of the "awkward
45    /// zone" (T50-59). This is necessary for certain cases where one role has
46    /// two backgrounds.
47    pub stay_together: bool,
48    pub constraint: DeltaConstraint,
49}
50
51impl<'a> ToneDeltaPair<'a> {
52    /// Defaults: `stay_together = true`, `constraint = Exact`.
53    pub const fn new(role_a: DynamicColor<'a>, role_b: DynamicColor<'a>, delta: f64, polarity: TonePolarity) -> Self {
54        Self {
55            role_a,
56            role_b,
57            delta,
58            polarity,
59            stay_together: true,
60            constraint: DeltaConstraint::Exact,
61        }
62    }
63
64    #[must_use]
65    pub const fn with_stay_together(mut self, stay_together: bool) -> Self {
66        self.stay_together = stay_together;
67        self
68    }
69
70    #[must_use]
71    pub const fn with_constraint(mut self, constraint: DeltaConstraint) -> Self {
72        self.constraint = constraint;
73        self
74    }
